from werkzeug.security import generate_password_hash @app.route( "/register", methods=["GET","POST"] ) def register(): if request.method == "POST": hashed_password = generate_password_hash( request.form["password"] ) user = User( username=request.form["username"], email=request.form["email"], password=hashed_password ) db.session.add(user) db.session.commit() return redirect("/login") return render_template("register.html")
